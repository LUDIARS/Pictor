cmake_minimum_required(VERSION 3.20)
project(PictorAndroidDemos LANGUAGES CXX)
if(NOT ANDROID)
    message(FATAL_ERROR "Use the Android NDK toolchain")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(PICTOR_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(PICTOR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PICTOR_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(PICTOR_ENABLE_SHARC OFF CACHE BOOL "" FORCE)
set(PICTOR_ENABLE_POSTPROCESS OFF CACHE BOOL "" FORCE)
set(PICTOR_ENABLE_DECALS OFF CACHE BOOL "" FORCE)
set(PICTOR_ENABLE_RIVE OFF CACHE BOOL "" FORCE)
add_subdirectory(../.. pictor-core)
add_library(pictor_demos SHARED native/host.cpp native/spheres_demo.cpp native/surface_demo.cpp)
target_link_libraries(pictor_demos PRIVATE pictor android log)
target_link_options(pictor_demos PRIVATE -Wl,--no-undefined)
